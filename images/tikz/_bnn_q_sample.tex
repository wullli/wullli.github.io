% Shared by bnn_q_sample.tex and bnn_q_sample_diabetes.tex.
% \bnnqsample{x1 name}{x2 name}{y name}: left, the learned q with one normal per
% weight; right, one sample from it. Empty names leave the feature labels out.

% \bnn{x offset}{prefix}: network with coordinates <prefix>1..15 on its weights
% (1..5 leave x1, 6..10 leave x2, 11..15 enter y)
\newcommand{\bnn}[5]{%
  \begin{scope}[xshift=#1*0.75pt]
    \node[io] (#2x1) at (40, 145) {$x_1$};
    \node[io] (#2x2) at (40, 255) {$x_2$};
    \node[io] (#2y)  at (455, 200) {$y$};
    \node[text=muted, font=\fsSmall] at (40, 110) {#3};
    \node[text=muted, font=\fsSmall] at (40, 291) {#4};
    \node[text=muted, font=\fsSmall] at (455, 236) {#5};
    \foreach \j in {1,...,5} {
      \node[hidden] (#2h\j) at (265, {80*\j - 40}) {};
      \pgfmathtruncatemacro{\k}{\j + 5}
      \pgfmathtruncatemacro{\m}{\j + 10}
      \draw[edge] (#2x1) -- coordinate[pos=.55] (#2\j) (#2h\j);
      \draw[edge] (#2x2) -- coordinate[pos=.84] (#2\k) (#2h\j);
      \draw[edge] (#2h\j) -- coordinate[pos=.5] (#2\m) (#2y);
    }
  \end{scope}}

\newcommand{\bnnqsample}[3]{%
\begin{tikzpicture}[px]
  \bnn{0}{q}{#1}{#2}{#3}
  % weight / mean / standard deviation of q
  \foreach \i/\mu/\sigma in {1/-0.1/0.95, 2/0.9/1, 3/1.9/0.22, 4/-0.2/0.85, 5/-1.1/0.25,
                             6/-0.3/0.85, 7/0.3/0.9, 8/-0.3/0.25, 9/-0.2/0.7, 10/-1.5/0.35,
                             11/0.1/0.35, 12/0.15/0.13, 13/1.2/0.13, 14/0.1/0.38, 15/-1.2/0.13}
    \gauss[mq]{($(q\i) + (0, 5)$)}{\mu}{\sigma}{18}{15.4};

  \draw[ink, line width=1.5pt, -{Triangle[length=9pt, width=7pt]}] (490, 200) -- (602, 200)
    node[midway, above=3pt] {sample}
    node[midway, below=3pt] {$\theta \sim \cq{q(\theta)}$};

  \bnn{605}{s}{#1}{#2}{#3}
  \foreach \i/\v in {1/0.6, 2/\textminus0.4, 3/1.2, 4/0.1, 5/\textminus0.9,
                     6/\textminus0.2, 7/0.3, 8/\textminus0.4, 9/\textminus0.8, 10/\textminus0.9,
                     11/0.3, 12/0.0, 13/0.8, 14/0.5, 15/\textminus0.8}
    \node[fill=white, inner sep=2pt, font=\fsSmall] at (s\i) {\v};

  \node[caption] at (265, 406) {Learned $\cq{q(\theta)}$: one $\mathcal{N}(\cpar{\mu_i}, \cpar{\sigma_i^2})$ per weight};
  \node[caption] at (870, 406) {One sample: an ordinary network};
\end{tikzpicture}}
