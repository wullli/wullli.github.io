% \gauss{(x,y)}{mu}{sigma}{half width}{peak}: N(mu, sigma^2) over [-3, 3]
% stretched to +-half width (px), above a baseline at (x,y) with a tick at 0.
% The peak is `peak`/sigma px tall, capped at 26px. A white backing hides
% the edge the glyph sits on.
\newcommand{\gauss}[5]{%
  \begin{scope}[shift={#1}]
    \pgfmathsetmacro{\gaussh}{min(26, #5/#3)}
    \fill[white] (-#4-4, 6) rectangle (#4+4, -\gaussh-2);
    \draw[edge, line width=0.75pt] (-#4, 0) -- (#4, 0) (0, 0) -- (0, 4);
    \draw[accent, line width=1.35pt, domain=-3:3, samples=241]
      plot ({\x*#4/3}, {-\gaussh*exp(-(\x-(#2))^2/(2*(#3)^2))});
  \end{scope}}
